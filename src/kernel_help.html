    I'll show a minimal, modern-friendly workflow to:

    • Build a 32-bit freestanding kernel in C with an assembly entry stub that calls kmain.
    • Compile and link the kernel into an ELF file you can boot with GRUB/qemu.
    • Implement a minimal Interrupt Descriptor Table (IDT) with one example handler (divide-by-zero).

    This is a raw kernel (not Linux). The toolchain used: GCC (i686), GNU ld, NASM/AS), and QEMU for testing.

    What we'll produce

    • entry.S — assembly stub with a Multiboot2-compatible header and a small setup (stack, call to kmain).
    • kernel.c — freestanding C kernel with kmain, VGA text helpers, and IDT setup.
    • isr_stub.S — small assembly ISR stub to push registers and call a C handler.
    • linker.ld — linker script to place sections properly.
    • Makefile and test instructions (QEMU).

    1. entry.S — multiboot2 header, basic entry that sets up stack and calls kmain (NASM syntax)
    vbnet

    ; entry.S - NASM syntax
; Multiboot2 header + simple entry that jumps to protected 32-bit code
BITS 32
GLOBAL multiboot_header
GLOBAL _start

; Multiboot2 header (minimal). GRUB will load the kernel into memory and jump to _start.
section .multiboot
align 8
multiboot_header:
    dd 0xE85250D6          ; magic
    dd 0x00000000          ; architecture (0 = i386)
    dd header_length       ; header length (filled below)
    dd header_checksum     ; checksum

    ; tag: end tag
    dd 0                   ; tag type = end
    dd 8                   ; size = 8

header_length:
    ; compute length from multiboot_header to end tag (8 + header content)
    ; NASM: we can compute by using labels difference at assemble time
    ; but for minimal header we can hardcode 16 (magic+arch+len+chksum = 16) + 8 tag = 24
    ; however above we used label; for simplicity compute value here:
    ; (if assembler doesn't allow computed dd, replace with dd 24)
    dd 24

header_checksum:
    ; checksum so that magic + arch + length + checksum = 0
    dd -(0xE85250D6 + 0 + 24)

section .text
align 4
_start:
    cli                     ; disable interrupts
    xor ebp, ebp
    mov esp, stack_top      ; set up stack
    call kernel_entry_c     ; call the C entry point
.hang:
    hlt
    jmp .hang

; import C symbol
extern kernel_entry_c

section .bss
align 16
stack_bottom:
    resb 16384              ; 16 KiB stack
stack_top:

    Notes:

    • If you prefer GAS/AT&T syntax instead of NASM, let me know and I’ll adapt.
    • This assumes GRUB loads the ELF; the Multiboot2 header is minimal to satisfy GRUB2.
    1. kernel.c — minimal kernel, VGA text printing, IDT setup and handler
    c

    /* kernel.c - freestanding 32-bit kernel */
#include <stdint.h>

typedef uint8_t  u8;
typedef uint16_t u16;
typedef uint32_t u32;
typedef uint64_t u64;

void kernel_putc(char c);
void kernel_puts(const char *s);

void idt_init(void);
void lidt(void *ptr);

void isr_common_handler(u32 vec, u32 errcode);

/* The symbol called from entry.S */
void kernel_entry_c(void) {
    kernel_puts("kmain: entered\n");

    idt_init();
    kernel_puts("IDT installed\n");

    /* Force a divide by zero to test ISR 0 (optional) */
    // volatile int x = 1 / 0;

    kernel_puts("kmain: done\n");
    for(;;) __asm__ __volatile__("hlt");
}

/* VGA text-mode helpers (80x25 color text) */
static volatile u16 *vga = (u16 *)0xB8000;
static int cursor_pos = 0;

void kernel_putc(char c) {
    if (c == '\n') {
        cursor_pos += 80 - (cursor_pos % 80);
        return;
    }
    vga[cursor_pos++] = (u16)c | (u16)0x0700;
}

void kernel_puts(const char *s) {
    while (*s) kernel_putc(*s++);
}

/* IDT structures */
struct idt_entry {
    u16 offset_low;
    u16 selector;
    u8  zero;
    u8  type_attr;
    u16 offset_high;
} __attribute__((packed));

struct idtr {
    u16 limit;
    u32 base;
} __attribute__((packed));

#define IDT_ENTRIES 256
static struct idt_entry idt[IDT_ENTRIES];

static void set_idt_entry(int n, void (*handler)(), u16 sel, u8 flags) {
    uintptr_t addr = (uintptr_t)handler;
    idt[n].offset_low  = addr & 0xFFFF;
    idt[n].selector    = sel;
    idt[n].zero        = 0;
    idt[n].type_attr   = flags;
    idt[n].offset_high = (addr >> 16) & 0xFFFF;
}

/* Extern ISR stub generated in assembly */
extern void isr0_stub(void);

/* Initialize a minimal IDT with ISR 0 (divide-by-zero) */
void idt_init(void) {
    for (int i = 0; i < IDT_ENTRIES; ++i) {
        set_idt_entry(i, 0, 0, 0);
    }

    /* 0x08 is typical kernel code segment selector in GDT when using GRUB */
    set_idt_entry(0, isr0_stub, 0x08, 0x8E); /* present, DPL=0, 32-bit interrupt gate */

    struct idtr idtr;
    idtr.limit = sizeof(idt) - 1;
    idtr.base  = (u32)&idt;
    lidt(&idtr);
}

/* Call from assembly to load IDTR */
void lidt(void *ptr) {
    __asm__ __volatile__("lidtl (%0)" : : "r"(ptr));
}

/* C handler called by assembly stub */
void isr_common_handler(u32 vec, u32 errcode) {
    kernel_puts("ISR handler: vec=");
    /* crude integer print */
    char buf[16];
    int i = 0;
    if (vec == 0) {
        kernel_puts("0 (divide by zero)\n");
    } else {
        kernel_puts("unknown\n");
    }
    (void)errcode;
}
    1. isr_stub.S — ISR assembly stub (NASM) that preserves registers and calls the C handler
    csharp

    ; isr_stub.S - NASM syntax for ISR0
BITS 32
GLOBAL isr0_stub
extern isr_common_handler

isr0_stub:
    cli
    pushad                  ; push general-purpose registers (eax..edi)
    push    0               ; error code (none for divide-by-zero, push 0)
    push    0               ; interrupt vector number (0)
    call    isr_common_handler
    add     esp, 8
    popad
    sti
    iretd

    1. Makefile — build targets (uses NASM and i686-elf-gcc or gcc -m32)
    ruby

    # Makefile
AS=nasm
CC=gcc
LD=ld
CFLAGS=-m32 -nostdlib -ffreestanding -fno-builtin -O2 -Wall -Wextra
ASFLAGS=-f elf32
LDFLAGS=-m elf_i386

OBJS=entry.o kernel.o isr_stub.o

all: kernel.bin

entry.o: entry.S
	$(AS) $(ASFLAGS) entry.S -o entry.o

isr_stub.o: isr_stub.S
	$(AS) $(ASFLAGS) isr_stub.S -o isr_stub.o

kernel.o: kernel.c
	$(CC) $(CFLAGS) -c kernel.c -o kernel.o

kernel.bin: $(OBJS)
	$(LD) $(LDFLAGS) -T linker.ld -o kernel.elf $(OBJS)
	# For GRUB, produce a multiboot-compatible ELF (kernel.elf is fine)
	# Optionally create a bootable ISO with grub if you want to test in qemu:
	mkdir -p iso/boot/grub
	cp kernel.elf iso/boot/kernel.elf
	printf 'set timeout=0\nset default=0\nmenuentry "mykernel" { multiboot /boot/kernel.elf }\n' > iso/boot/grub/grub.cfg
	grub-mkrescue -o kernel.iso iso

run: kernel.bin
	qemu-system-i386 -cdrom kernel.iso

clean:
	rm -f *.o kernel.elf kernel.iso
	rm -rf iso

    Notes:

    • On many systems you need multiboot-compliant ELF and use grub-mkrescue to build an ISO.
    • If your toolchain uses i686-elf-gcc/i686-elf-ld adjust CC/LD accordingly.
    • -m32 is essential if your host is 64-bit and you want i386 code. You may need libc6-dev-i386 or cross-toolchain.

    2) Explanation / rationale (brief)

    • Use a tiny assembly entry to set up a stack and call a C entry (kernel_entry_c). This keeps C code simple and portable.
    • Use Multiboot2 header so GRUB loads your ELF and provides a clean environment (GDT for 32-bit protected mode, etc.). Alternatively, if you already have your own 32-bit bootloader that enters protected mode, replace the multiboot header with your own stub.
    • Build as a freestanding binary: -ffreestanding -nostdlib -fno-builtin.
    • For IDT: create an array of 256 entries, fill entries with the offset and flags, then lidt the IDTR. Use an ISR stub in assembly that preserves registers and then calls a C handler.
    • For hardware: code segment selector 0x08 is used because GRUB typically sets up a GDT where the kernel code selector is at that offset. If you write your own bootloader/GDT, ensure the selector matches.

    3) Testing

    • Build: make
    • Run in QEMU (Makefile target): make run (requires grub-mkrescue and QEMU).
    • Or manually:
      • grub-mkrescue -o kernel.iso iso
      • qemu-system-i386 -cdrom kernel.iso

    You should see your kernel text printed via VGA.

    4) Tips & alternatives

    • If you already have a boot sector & GDT that enters protected mode, you can skip the Multiboot header and make entry.S match your loader expectations.
    • For more ISRs: create an array of stubs (isr0..isr31), set each IDT entry, and provide common error-code handling.
    • Consider writing a small gdt.s or using GRUB-provided GDT; if you implement your own GDT, make sure segment selectors (like 0x08) match your GDT layout.

    What does an ISR stub do?

    An ISR (interrupt service routine) stub is the small assembly function that runs immediately on an interrupt/exception. Its job is to:

    • Preserve CPU state (push registers) so the interrupted code can resume safely.
    • Push an interrupt/vector number and/or an error code (if hardware/exception provides one) and then call a higher-level C handler.
    • Restore registers and return from interrupt with iret/iretd.
    • Optionally mask/enable interrupts (cli/sti) and handle EOI for PIC/APIC.

    Shortly:

    1. Go to the IVT (interrupt vector table) and modify it with your ISR segment and offset, saving the old values
    2. ... your ISR should be re-entrant ...
    3. remember to call the old ISR after you're done with the work (this is called interrupt chaining)
    4. your ISR should be re-entrant (so that if the interrupt is fired again when your ISR is still running the pc won't explode)

    Concrete minimal ISR stub + C handler (recommended pattern)

    • Stub (assembly) must save registers, push vector & error code (or a fake one), call C handler, restore, iretd.

BITS 32

isr0_stub:
    cli                 ; disable interrupts while handling
    pushad              ; push eax..edi
    push    0           ; error code (exceptions without error code)
    push    0           ; vector number (0 = divide-by-zero)
    call    isr_common_handler
    add     esp, 8      ; pop the two pushed args
    popad               ; restore eax..edi
    sti                 ; re-enable interrupts if desired
    iretd               ; return from interrupt (32-bit)

    /* Minimal: print or handle the vector */
    /* Be careful: printing may use registers; ensure it's safe or re-entrant */
    (void)errcode;
    if (vec == 0) {
        /* handle divide-by-zero */
    }
}

    Minimal kernel.c (freestanding, no IDT/ISR) — prints text to VGA

    Save as kernel.c. This is a smallest practical freestanding C kmain that writes a short string to VGA text memory and then halts.

    c

    /* kernel.c - minimal freestanding kernel (32-bit) */
#include <stdint.h>

typedef unsigned int u32;

/* C entry called from assembly */
void kmain(void) {
    volatile uint16_t *vga = (uint16_t *)0xB8000;
    const char *s = "Hello from kmain!";
    /* simple orange-on-black attribute 0x0E (you can pick any) */
    uint16_t attr = 0x0E << 8;

    for (int i = 0; s[i]; ++i) {
        vga[i] = (uint16_t)s[i] | attr;
    }

    /* halt forever */
    for (;;) {
        __asm__ volatile ("cli; hlt");
    }
}
